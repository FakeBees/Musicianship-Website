```python
class Container(db.Model):
    __tablename__ = 'container'
    id          = db.Column(db.Integer, primary_key=True)
    name        = db.Column(db.String(120), nullable=False, unique=True)
    description = db.Column(db.String(300))
    created_at  = db.Column(db.DateTime, server_default=db.func.now())
    melodies    = db.relationship('Melody', backref='container', lazy='dynamic')

    def __repr__(self):
        return f'<Container {self.name}>'


class GenProgression(db.Model):
    __tablename__ = 'gen_progression'
    id          = db.Column(db.Integer, primary_key=True)
    number      = db.Column(db.Integer, unique=True, nullable=False)
    name        = db.Column(db.String(120))
    length_bars = db.Column(db.Integer, nullable=False, default=4)
    mode        = db.Column(db.String(10), nullable=False, default='major')
    chords_json = db.Column(db.Text, nullable=False)
    difficulty  = db.Column(db.Integer, nullable=False, default=1)

    @property
    def chords(self):
        return json.loads(self.chords_json)

    def __repr__(self):
        return f'<GenProgression #{self.number} {self.name}>'
```

```python
        # --- Add new columns if missing (SQLite requires ALTER TABLE) ---
        import sqlite3
        db_path = os.path.join(os.path.dirname(__file__), 'instance', 'musicianship.db')
        conn = sqlite3.connect(db_path)
        cur = conn.cursor()

        def add_column(table, col, col_type):
            cur.execute(f"PRAGMA table_info({table})")
            cols = [r[1] for r in cur.fetchall()]
            if col not in cols:
                cur.execute(f"ALTER TABLE {table} ADD COLUMN {col} {col_type}")
                print(f"  Added {table}.{col}")
            else:
                print(f"  {table}.{col} already exists")

        add_column('melody', 'container_id', 'INTEGER')
        add_column('melody', 'generation_params', 'TEXT')
        add_column('chord_progression', 'public_id', 'VARCHAR(12)')
        add_column('rhythm', 'public_id', 'VARCHAR(12)')
        add_column('holistic_exercise', 'public_id', 'VARCHAR(12)')
        conn.commit()
        conn.close()

        # --- Backfill public_ids ---
```

```python
@app.route('/admin/melodies')
@login_required
@role_required('admin')
def admin_melodies():
    q = request.args.get('q', '').strip()
    tag_filter = request.args.get('tag', '').strip()
    container_filter = request.args.get('container', type=int)
    query = Melody.query
    if q:
        query = query.filter(db.or_(Melody.name.ilike(f'%{q}%'), Melody.public_id.ilike(f'%{q}%')))
    if tag_filter:
        query = query.filter(Melody.tags.any(Tag.name == tag_filter))
    if container_filter:
        query = query.filter_by(container_id=container_filter)
    melodies = query.order_by(Melody.id.desc()).all()
    all_tags = Tag.query.order_by(Tag.name).all()
    containers = Container.query.order_by(Container.name).all()
    return render_template('admin/melodies.html', melodies=melodies, all_tags=all_tags,
                           containers=containers, q=q, tag_filter=tag_filter,
                           container_filter=container_filter)


@app.route('/admin/melodies/<int:mel_id>/edit', methods=['GET', 'POST'])

    mel = Melody.query.get_or_404(mel_id)
    all_tags = Tag.query.order_by(Tag.name).all()

    if request.method == 'POST':
        mel.name          = request.form.get('name', '').strip() or mel.name
        mel.description   = request.form.get('description', '').strip()
        mel.key_signature = request.form.get('key_signature', mel.key_signature)

        mel.clef          = request.form.get('clef', mel.clef)
```

```python
# ── Melody MIDI Upload ───────────────────────────────────────────────────────

ALLOWED_MIDI = {'mid', 'midi'}

def _allowed_midi(filename):
    return '.' in filename and filename.rsplit('.', 1)[1].lower() in ALLOWED_MIDI


@app.route('/admin/melodies/upload', methods=['GET', 'POST'])
@login_required
@role_required('admin')
def admin_melody_upload():
    all_tags   = Tag.query.order_by(Tag.name).all()
    containers = Container.query.order_by(Container.name).all()

    if request.method == 'GET':
        return render_template('admin/melody_upload.html', all_tags=all_tags,
                               containers=containers, parsed=None)

    # POST: either parsing step or save step
    action = request.form.get('action', 'parse')

    if action == 'save':
        # Save the previously-parsed melody
        notes_json_str = request.form.get('notes_json', '').strip()
        midi_filename  = request.form.get('midi_filename', '').strip()
        name           = request.form.get('name', '').strip()
        if not name or not notes_json_str:
            flash('Name and notes are required.', 'danger')
            return redirect(url_for('admin_melody_upload'))

        # Determine public_id
        last = Melody.query.order_by(Melody.id.desc()).first()
        next_num = (last.id + 1) if last else 1
        public_id = f'MEL-{next_num:04d}'

        tag_ids = request.form.getlist('tag_ids', type=int)
        container_id = request.form.get('container_id', type=int) or None

        mel = Melody(
            name=name,
            description=request.form.get('description', '').strip(),
            midi_filename=midi_filename,
            notes_json=notes_json_str,
            key_signature=request.form.get('key_signature', 'C'),
            time_signature=request.form.get('time_signature', '4/4'),
            clef=request.form.get('clef', 'treble'),
            min_duration=request.form.get('min_duration', 'q'),
            tempo=request.form.get('tempo', 100, type=int),
            difficulty=request.form.get('difficulty', 1, type=int),
            public_id=public_id,
            container_id=container_id,
        )
        mel.tags = Tag.query.filter(Tag.id.in_(tag_ids)).all() if tag_ids else []
        db.session.add(mel)
        db.session.commit()
        flash(f'Melody "{name}" saved ({public_id}).', 'success')
        return redirect(url_for('admin_edit_melody', mel_id=mel.id))

    # action == 'parse': handle MIDI file upload
    file = request.files.get('midi_file')
    if not file or not file.filename or not _allowed_midi(file.filename):
        flash('Please upload a valid .mid file.', 'danger')
        return render_template('admin/melody_upload.html', all_tags=all_tags,
                               containers=containers, parsed=None)

    key = request.form.get('key_signature', 'C')
    filename = secure_filename(file.filename)
    tmp_path = os.path.join(app.static_folder, '_tmp_upload.mid')
    file.save(tmp_path)

    try:
        raw_notes  = extract_notes(tmp_path)
        notes_list = build_json_list(raw_notes, key)
        notes_json_str = json.dumps(notes_list)
        # Guess min_duration from what's in the notes
        dur_set = {n['duration'].rstrip('r') for n in notes_list}
        dur_order = ['16', '8', 'q', 'h', 'w']
        min_dur = next((d for d in dur_order if d in dur_set), 'q')
        # Derive slug for storage
        slug = os.path.splitext(filename)[0]
        dest_dir = os.path.join(app.static_folder, 'melodic', slug)
        os.makedirs(dest_dir, exist_ok=True)
        dest_path = os.path.join(dest_dir, filename)
        os.replace(tmp_path, dest_path)
        midi_filename = f'melodic/{slug}/{filename}'
        parsed = {
            'notes_json': notes_json_str,
            'midi_filename': midi_filename,
            'min_duration': min_dur,
            'note_count': len(notes_list),
        }
    except Exception as e:
        os.remove(tmp_path) if os.path.exists(tmp_path) else None
        flash(f'MIDI parse error: {e}', 'danger')
        parsed = None

    return render_template('admin/melody_upload.html', all_tags=all_tags,
                           containers=containers, parsed=parsed,
                           key_signature=key)
```

```python
    from chord_utils import infer_chords_from_midi

        return render_template('admin/harmonic_upload.html', all_tags=all_tags, parsed=None)

            name=name,
            description=request.form.get('description', '').strip(),
            midi_filename=midi_filename,
            chords_json=chords_json_str,
            key_signature=request.form.get('key_signature', 'C'),

            difficulty=request.form.get('difficulty', 1, type=int),
```

```python
@app.route('/admin/holistic/upload', methods=['GET', 'POST'])
@login_required
@role_required('admin')
def admin_holistic_upload():
    """
    Upload a holistic exercise: WAV + melody.mid + harmony.mid.
    Files go into static/holistic/<slug>/.
    Folder convention matches existing holistic exercises (see HOLISTIC_DICTATION_PLAN.md).
    """
    from chord_utils import infer_chords_from_midi
    all_tags = Tag.query.order_by(Tag.name).all()

    if request.method == 'GET':
        return render_template('admin/holistic_upload.html', all_tags=all_tags)

    # Collect files
    wav_file     = request.files.get('wav_file')
    melody_mid   = request.files.get('melody_mid')
    harmony_mid  = request.files.get('harmony_mid')
    name         = request.form.get('name', '').strip()
    key          = request.form.get('key_signature', 'C')

    if not name or not wav_file or not melody_mid:
        flash('Name, WAV file, and melody MIDI are required.', 'danger')
        return render_template('admin/holistic_upload.html', all_tags=all_tags)

    # Create folder
    slug = secure_filename(name.lower().replace(' ', '_'))
    folder = f'holistic/{slug}/'
    dest_dir = os.path.join(app.static_folder, 'holistic', slug)
    os.makedirs(dest_dir, exist_ok=True)

    # Save WAV
    wav_filename = secure_filename(wav_file.filename) if wav_file.filename else 'audio.wav'
    wav_file.save(os.path.join(dest_dir, wav_filename))

    # Parse melody MIDI
    mel_tmp = os.path.join(dest_dir, '_tmp_melody.mid')
    melody_mid.save(mel_tmp)
    try:
        raw = extract_notes(mel_tmp)
        melody_notes = build_json_list(raw, key)
        os.rename(mel_tmp, os.path.join(dest_dir, 'melody.mid'))
    except Exception as e:
        flash(f'Melody MIDI parse error: {e}', 'danger')
        return render_template('admin/holistic_upload.html', all_tags=all_tags)

    # Parse harmony MIDI (optional)
    harmony_chords = []
    if harmony_mid and harmony_mid.filename:
        har_tmp = os.path.join(dest_dir, '_tmp_harmony.mid')
        harmony_mid.save(har_tmp)
        try:
            harmony_chords = infer_chords_from_midi(har_tmp, key_signature=key)
            os.rename(har_tmp, os.path.join(dest_dir, 'harmony.mid'))
        except Exception as e:
            flash(f'Harmony MIDI parse error (skipped): {e}', 'warning')

    # Determine next public_id
    last = HolisticExercise.query.order_by(HolisticExercise.id.desc()).first()
    next_num = (last.id + 1) if last else 1
    public_id = f'HOL-{next_num:04d}'

    tag_ids = request.form.getlist('tag_ids', type=int)
    ex = HolisticExercise(
        name=name,
        description=request.form.get('description', '').strip(),
        folder=folder,
        wav_filename=wav_filename,
        key_signature=key,
        time_signature=request.form.get('time_signature', '4/4'),
        tempo=request.form.get('tempo', 120, type=int),
        difficulty=request.form.get('difficulty', 1, type=int),
        melody_clef=request.form.get('melody_clef', 'treble'),
        is_major=(request.form.get('is_major', 'true') == 'true'),
        melody_notes_json=json.dumps(melody_notes),
        harmony_chords_json=json.dumps(harmony_chords),
        extra_lines_json='[]',
        public_id=public_id,
    )
    ex.tags = Tag.query.filter(Tag.id.in_(tag_ids)).all() if tag_ids else []
    db.session.add(ex)
    db.session.commit()
    flash(f'Holistic exercise "{name}" saved ({public_id}).', 'success')
    return redirect(url_for('admin_edit_holistic', ex_id=ex.id))
```
